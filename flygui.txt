local function GiveWeapon(weaponName)
    local found = false
    for _, v in pairs(game:GetDescendants()) do
        if v:IsA("Tool") and v.Name:lower() == weaponName:lower() then
            local clone = v:Clone()
            clone.Parent = Player.Backpack
            found = true
            print("Оружие получено: " .. v.Name)
            break
        end
    end
    if not found then print("Оружие не найдено в этой игре!") end
end

    btn.TextColor3 = Color3.new(0, 1, 1)
    btn.MouseButton1Click:Connect(function() GiveWeapon(name) end)
end

local function AddTrollBtn(name, callback)
    local btn = Instance.new("TextButton", TrollsFrame)
    btn.Size = UDim2.new(0.9, 0, 0, 35)
    btn.Text = name
    btn.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
    btn.TextColor3 = Color3.new(1, 1, 1)
    btn.MouseButton1Click:Connect(function() callback(btn) end)
end

AddTrollBtn("Fly: OFF (F)", function(b)
    States.Fly = not States.Fly
    b.Text = "Fly: " .. (States.Fly and "ON" or "OFF")
end)

AddTrollBtn("Speed: 16", function(b)
    if States.Speed == 16 then States.Speed = 100 else States.Speed = 16 end
    Player.Character.Humanoid.WalkSpeed = States.Speed
    b.Text = "Speed: " .. States.Speed
end)

AddTrollBtn("Invisible Fling: OFF", function(b)
    States.Fling = not States.Fling
    States.Invisible = States.Fling
    b.Text = "Invisible Fling: " .. (States.Fling and "ON" or "OFF")
    
    for _, v in pairs(Player.Character:GetDescendants()) do
        if v:IsA("BasePart") or v:IsA("Decal") then
            v.Transparency = States.Invisible and 1 or 0
        end
    end
end)

AddTrollBtn("Jerk: OFF", function(b)
    States.Jerk = not States.Jerk
    b.Text = "Jerk: " .. (States.Jerk and "ON" or "OFF")
end)

RunService.Stepped:Connect(function()
    local char = Player.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    
    if States.Fly then
        local hrp = char.HumanoidRootPart
        local vel = hrp:FindFirstChild("FlyVel") or Instance.new("BodyVelocity", hrp)
        vel.Name = "FlyVel"
        vel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        
        local moveDir = char.Humanoid.MoveDirection
        local flyVel = moveDir * 100
        if UIS:IsKeyDown(Enum.KeyCode.Space) then flyVel = flyVel + Vector3.new(0, 50, 0)
        elseif UIS:IsKeyDown(Enum.KeyCode.LeftShift) then flyVel = flyVel - Vector3.new(0, 50, 0) end
        vel.Velocity = flyVel
        
        hrp.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + Mouse.LookVector)
    else
        if char.HumanoidRootPart:FindFirstChild("FlyVel") then char.HumanoidRootPart.FlyVel:Destroy() end
    end

    if States.Fling then
        char.HumanoidRootPart.Velocity = Vector3.new(999999, 999999, 999999)
        char.HumanoidRootPart.RotVelocity = Vector3.new(999999, 999999, 999999)
    end

    if States.Jerk then
        char.Humanoid.RootPart.RootJoint.C0 = char.Humanoid.RootPart.RootJoint.C0 * CFrame.Angles(math.rad(math.random(-15,15)), 0, 0)
    end
end)

UIS.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if input.KeyCode == MenuKey then
        ScreenGui.Enabled = not ScreenGui.Enabled
    elseif input.KeyCode == FlyKey then
        States.Fly = not States.Fly
    end
end)
